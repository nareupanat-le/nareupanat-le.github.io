\section{Limits of Functions}

\subsection{Type 1.1: Factoring ($0/0$ Quadratic / Difference of Squares)}
\begin{problembox}{Problem 1.1: Limit by Factoring}
Evaluate the limit:
\[
\lim_{x \to 2} \frac{x^2 - 4}{x - 2}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form by direct substitution:
	\begin{equation*}
		\frac{2^2 - 4}{2 - 2} = \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Factor the numerator using the difference of squares identity $a^2 - b^2 = (a - b)(a + b)$:
	\begin{equation*}
		x^2 - 4 = (x - 2)(x + 2).
	\end{equation*}

	\item [Step 3.] Simplify the rational function by cancelling the common non-zero factor $(x - 2)$ for $x \neq 2$:
	\begin{equation*}
		\frac{x^2 - 4}{x - 2} = \frac{(x - 2)(x + 2)}{x - 2} = x + 2.
	\end{equation*}

	\item [Step 4.] Evaluate the limit by direct substitution:
	\begin{equation*}
		\lim_{x \to 2} \frac{x^2 - 4}{x - 2} = \lim_{x \to 2} (x + 2) = 2 + 2 = 4.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.2: Square Root Conjugate ($0/0$)}
\begin{problembox}{Problem 1.2: Limit by Conjugate (Square Root)}
Evaluate the limit:
\[
\lim_{x \to 0} \frac{\sqrt{x + 4} - 2}{x}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form by direct substitution:
	\begin{equation*}
		\frac{\sqrt{0 + 4} - 2}{0} = \frac{2 - 2}{0} = \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Multiply the numerator and denominator by the conjugate $(\sqrt{x + 4} + 2)$:
	\begin{equation*}
		\frac{\sqrt{x + 4} - 2}{x} \cdot \frac{\sqrt{x + 4} + 2}{\sqrt{x + 4} + 2} = \frac{(\sqrt{x + 4})^2 - 2^2}{x(\sqrt{x + 4} + 2)}.
	\end{equation*}

	\item [Step 3.] Simplify the expression:
	\begin{equation*}
		\frac{(x + 4) - 4}{x(\sqrt{x + 4} + 2)} = \frac{x}{x(\sqrt{x + 4} + 2)} = \frac{1}{\sqrt{x + 4} + 2}.
	\end{equation*}

	\item [Step 4.] Evaluate the limit by direct substitution $x = 0$:
	\begin{equation*}
		\lim_{x \to 0} \frac{\sqrt{x + 4} - 2}{x} = \lim_{x \to 0} \frac{1}{\sqrt{x + 4} + 2} = \frac{1}{\sqrt{0 + 4} + 2} = \frac{1}{2 + 2} = \frac{1}{4}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.3: Cube Root Conjugate / Difference of Cubes ($0/0$)}
\begin{problembox}{Problem 1.3: Limit by Cube Root Conjugate}
Evaluate the limit:
\[
\lim_{x \to 0} \frac{\sqrt[3]{x + 8} - 2}{x}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form by direct substitution:
	\begin{equation*}
		\frac{\sqrt[3]{0 + 8} - 2}{0} = \frac{2 - 2}{0} = \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Use the identity $(a - b)(a^2 + ab + b^2) = a^3 - b^3$. Multiply numerator and denominator by $(\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4$:
	\begin{equation*}
		\frac{\sqrt[3]{x + 8} - 2}{x} \cdot \frac{(\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4}{(\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4} = \frac{(x + 8) - 8}{x\left((\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4\right)}.
	\end{equation*}

	\item [Step 3.] Simplify the expression:
	\begin{equation*}
		\frac{x}{x\left((\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4\right)} = \frac{1}{(\sqrt[3]{x + 8})^2 + 2\sqrt[3]{x + 8} + 4}.
	\end{equation*}

	\item [Step 4.] Evaluate the limit by direct substitution $x = 0$:
	\begin{equation*}
		\lim_{x \to 0} \frac{1}{(\sqrt[3]{8})^2 + 2\sqrt[3]{8} + 4} = \frac{1}{4 + 4 + 4} = \frac{1}{12}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.4: Complex Fraction ($0/0$)}
\begin{problembox}{Problem 1.4: Limit of Complex Fractions}
Evaluate the limit:
\[
\lim_{x \to 0} \frac{\frac{1}{x + 3} - \frac{1}{3}}{x}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form by direct substitution:
	\begin{equation*}
		\frac{\frac{1}{0 + 3} - \frac{1}{3}}{0} = \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Combine the terms in the numerator with a common denominator $3(x + 3)$:
	\begin{equation*}
		\frac{1}{x + 3} - \frac{1}{3} = \frac{3 - (x + 3)}{3(x + 3)} = \frac{-x}{3(x + 3)}.
	\end{equation*}

	\item [Step 3.] Simplify the complex fraction:
	\begin{equation*}
		\frac{\frac{-x}{3(x + 3)}}{x} = \frac{-x}{3x(x + 3)} = -\frac{1}{3(x + 3)}.
	\end{equation*}

	\item [Step 4.] Evaluate the limit by direct substitution $x = 0$:
	\begin{equation*}
		\lim_{x \to 0} -\frac{1}{3(x + 3)} = -\frac{1}{3(0 + 3)} = -\frac{1}{9}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.5: Double Conjugate (Numerator \& Denominator)}
\begin{problembox}{Problem 1.5: Double Conjugate Limit}
Evaluate the limit:
\[
\lim_{x \to 0} \frac{\sqrt{x + 4} - 2}{\sqrt{x + 9} - 3}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form by direct substitution:
	\begin{equation*}
		\frac{\sqrt{4} - 2}{\sqrt{9} - 3} = \frac{2 - 2}{3 - 3} = \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Multiply by both conjugates $(\sqrt{x + 4} + 2)$ and $(\sqrt{x + 9} + 3)$:
	\begin{equation*}
		\frac{\sqrt{x + 4} - 2}{\sqrt{x + 9} - 3} \cdot \frac{\sqrt{x + 4} + 2}{\sqrt{x + 4} + 2} \cdot \frac{\sqrt{x + 9} + 3}{\sqrt{x + 9} + 3}
		= \frac{((x + 4) - 4)(\sqrt{x + 9} + 3)}{((x + 9) - 9)(\sqrt{x + 4} + 2)}.
	\end{equation*}

	\item [Step 3.] Simplify and cancel common factor $x$:
	\begin{equation*}
		\frac{x(\sqrt{x + 9} + 3)}{x(\sqrt{x + 4} + 2)} = \frac{\sqrt{x + 9} + 3}{\sqrt{x + 4} + 2}.
	\end{equation*}

	\item [Step 4.] Evaluate the limit by direct substitution $x = 0$:
	\begin{equation*}
		\lim_{x \to 0} \frac{\sqrt{x + 9} + 3}{\sqrt{x + 4} + 2} = \frac{\sqrt{9} + 3}{\sqrt{4} + 2} = \frac{3 + 3}{2 + 2} = \frac{6}{4} = \frac{3}{2}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.6: Basic Trigonometric Limits ($\frac{\sin kx}{x}$, $\frac{\tan kx}{\sin mx}$)}
\begin{problembox}{Problem 1.6: Trigonometric Limit}
Evaluate the limit:
\[
\lim_{x \to 0} \frac{\tan(3x)}{\sin(5x)}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and standard trigonometric limits:
	\begin{align*}
		\lim_{u \to 0} \frac{\sin(u)}{u} &= 1 \\
		\lim_{u \to 0} \frac{\tan(u)}{u} &= 1
	\end{align*}
	
	\item [Step 2.] Multiply and divide by $3x$ and $5x$ to obtain standard limit forms:
	\begin{equation*}
		\frac{\tan(3x)}{\sin(5x)} = \frac{\frac{\tan(3x)}{3x} \cdot 3x}{\frac{\sin(5x)}{5x} \cdot 5x} = \frac{3}{5} \cdot \frac{\frac{\tan(3x)}{3x}}{\frac{\sin(5x)}{5x}}.
	\end{equation*}

	\item [Step 3.] Evaluate the limit as $x \to 0$:
	\begin{equation*}
		\lim_{x \to 0} \frac{\tan(3x)}{\sin(5x)} = \frac{3}{5} \cdot \frac{\lim_{x \to 0}\frac{\tan(3x)}{3x}}{\lim_{x \to 0}\frac{\sin(5x)}{5x}} = \frac{3}{5} \cdot \frac{1}{1} = \frac{3}{5}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.7: Limits at Infinity ($\infty/\infty$ \& $\infty - \infty$)}
\begin{problembox}{Problem 1.7: Limit at Infinity}
Evaluate the limit:
\[
\lim_{x \to \infty} \left( \sqrt{x^2 + 4x} - x \right)
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Check the indeterminate form: $\infty - \infty$.
	
	\item [Step 2.] Multiply and divide by the conjugate $(\sqrt{x^2 + 4x} + x)$:
	\begin{equation*}
		\frac{(\sqrt{x^2 + 4x} - x)(\sqrt{x^2 + 4x} + x)}{\sqrt{x^2 + 4x} + x} = \frac{(x^2 + 4x) - x^2}{\sqrt{x^2 + 4x} + x} = \frac{4x}{\sqrt{x^2 + 4x} + x}.
	\end{equation*}

	\item [Step 3.] Divide numerator and denominator by $x$ (noting $\sqrt{x^2} = x$ for $x > 0$):
	\begin{equation*}
		\frac{\frac{4x}{x}}{\sqrt{\frac{x^2 + 4x}{x^2}} + \frac{x}{x}} = \frac{4}{\sqrt{1 + \frac{4}{x}} + 1}.
	\end{equation*}

	\item [Step 4.] Evaluate the limit as $x \to \infty$, since $\lim_{x \to \infty} \frac{4}{x} = 0$:
	\begin{equation*}
		\lim_{x \to \infty} \frac{4}{\sqrt{1 + \frac{4}{x}} + 1} = \frac{4}{\sqrt{1 + 0} + 1} = \frac{4}{2} = 2.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.8: One-Sided Limit with Absolute Value}
\begin{problembox}{Problem 1.8: One-Sided Limit with $|x|$}
Evaluate the one-sided limit:
\[
\lim_{x \to 2^-} \frac{|x - 2|}{x^2 - 4}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Apply the piecewise definition of the absolute value function:
	\begin{equation*}
		|x - 2| = \begin{cases} x - 2, & \text{if } x \ge 2 \\ -(x - 2), & \text{if } x < 2 \end{cases}
	\end{equation*}
	Since $x \to 2^-$, we have $x < 2$, which implies $|x - 2| = -(x - 2)$.

	\item [Step 2.] Factor the denominator: $x^2 - 4 = (x - 2)(x + 2)$.
	
	\item [Step 3.] Substitute and simplify the expression for $x < 2$:
	\begin{equation*}
		\frac{|x - 2|}{x^2 - 4} = \frac{-(x - 2)}{(x - 2)(x + 2)} = -\frac{1}{x + 2}.
	\end{equation*}

	\item [Step 4.] Evaluate the one-sided limit:
	\begin{equation*}
		\lim_{x \to 2^-} \frac{|x - 2|}{x^2 - 4} = \lim_{x \to 2^-} \left(-\frac{1}{x + 2}\right) = -\frac{1}{2 + 2} = -\frac{1}{4}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 1.9: L'H\^opital's Rule ($0/0$ or $\infty/\infty$)}
\begin{problembox}{Problem 1.9: L'H\^opital's Rule}
Evaluate the limit using L'H\^opital's Rule:
\[
\lim_{x \to 0} \frac{e^{2x} - 1}{\sin(3x)}
\]
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Verify the indeterminate form:
	\begin{equation*}
		\lim_{x \to 0} (e^{2x} - 1) = 1 - 1 = 0 \quad \text{and} \quad \lim_{x \to 0} \sin(3x) = 0 \implies \frac{0}{0} \quad (\text{Indeterminate Form}).
	\end{equation*}
	
	\item [Step 2.] Apply L'H\^opital's Rule by differentiating numerator and denominator:
	\begin{equation*}
		\lim_{x \to 0} \frac{e^{2x} - 1}{\sin(3x)} = \lim_{x \to 0} \frac{\frac{d}{dx}[e^{2x} - 1]}{\frac{d}{dx}[\sin(3x)]}.
	\end{equation*}

	\item [Step 3.] Calculate the derivatives:
	\begin{itemize}
		\item $\frac{d}{dx}[e^{2x} - 1] = 2e^{2x}$.
		\item $\frac{d}{dx}[\sin(3x)] = 3\cos(3x)$.
	\end{itemize}

	\item [Step 4.] Evaluate the resulting limit:
	\begin{equation*}
		\lim_{x \to 0} \frac{2e^{2x}}{3\cos(3x)} = \frac{2e^0}{3\cos(0)} = \frac{2(1)}{3(1)} = \frac{2}{3}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\section{Basic Derivatives \& Term-by-Term Differentiation}

\subsection{Type 2.1: Polynomial with Fractional and Negative Powers}
\begin{problembox}{Problem 2.1: Powers and Radicals}
Given $f(x) = 3x^3 + \frac{3}{x} + 5\sqrt{x}$, find $f'(x)$ and evaluate $f'(1)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		u^n &= \frac{1}{u^{-n}} \\
		\sqrt[n]{u^m} &= u^{\frac{m}{n}} \\
		[u^n]' &= n u^{n-1} u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left(3x^3 + \frac{3}{x} + 5\sqrt{x}\right) \\
		&= \frac{d}{dx} 3x^3 + \frac{d}{dx} \frac{3}{x} + \frac{d}{dx} 5\sqrt{x} \\
		&= 3 \frac{d}{dx} x^3 + 3 \frac{d}{dx} x^{-1} + 5 \frac{d}{dx} x^{\frac{1}{2}} \\
		&= (3)(3)x^2 + (3)(-1)x^{-2} + (5)\left(\frac{1}{2}\right)x^{-\frac{1}{2}}.
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = 9x^2 - \frac{3}{x^2} + \frac{5}{2\sqrt{x}}
	\end{equation*}
	
	\item [Step 3.] Compute $f'(1)$ by putting $x = 1$ into $f'(x)$, then we have
	\begin{align*}
		f'(1) &= 9(1)^2 - \frac{3}{(1)^2} + \frac{5}{2\sqrt{1}} \\
		&= 9(1) - \frac{3}{1} + \frac{5}{2} \\
		&= 9 - 3 + \frac{5}{2} \\
		&= \frac{17}{2}.
	\end{align*}
	Therefore, $f'(1) = \frac{17}{2}$.
	\end{enumerate}
\end{solutionbox}

\subsection{Type 2.2: Basic Trigonometric Functions (Sine, Cosine, Secant, Tangent)}
\begin{problembox}{Problem 2.2: Trig Derivatives}
Given $f(x) = 4\sin(3x) - 2\cos(3x)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\sin(u)]' &= \cos(u) u' \\
		[\cos(u)]' &= -\sin(u) u' \\
		[u^n]' &= n u^{n-1} u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left(4\sin(3x) - 2\cos(3x)\right) \\
		&= \frac{d}{dx} 4\sin(3x) - \frac{d}{dx} 2\cos(3x) \\
		&= 4 \frac{d}{dx} \sin(3x) - 2 \frac{d}{dx} \cos(3x) \\
		&= 4\cos(3x)\frac{d}{dx}(3x) - 2(-\sin(3x))\frac{d}{dx}(3x) \\
		&= (4)\cos(3x)(3) - (2)(-\sin(3x))(3).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = 12\cos(3x) + 6\sin(3x)
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= 12\cos(3(0)) + 6\sin(3(0)) \\
		&= 12\cos(0) + 6\sin(0) \\
		&= 12(1) + 6(0) \\
		&= 12.
	\end{align*}
	Therefore, $f'(0) = 12$.
	\end{enumerate}
\end{solutionbox}

\subsection{Type 2.3: Natural and General Exponential Functions ($e^{kx}$ and $a^u$)}
\begin{problembox}{Problem 2.3: Exponential Derivatives}
Given $f(x) = 3^{2x+1} + 2e^{3x}$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[a^u]' &= a^u \ln(a) u' \\
		[e^u]' &= e^u u' \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left(3^{2x+1} + 2e^{3x}\right) \\
		&= \frac{d}{dx} 3^{2x+1} + \frac{d}{dx} 2e^{3x} \\
		&= \frac{d}{dx} 3^{2x+1} + 2 \frac{d}{dx} e^{3x} \\
		&= 3^{2x+1}\ln(3)\frac{d}{dx}(2x+1) + 2e^{3x}\frac{d}{dx}(3x) \\
		&= 3^{2x+1}\ln(3)(2) + (2)e^{3x}(3).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = 2\ln(3)3^{2x+1} + 6e^{3x}
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= 2\ln(3)3^{2(0)+1} + 6e^{3(0)} \\
		&= 2\ln(3)3^1 + 6e^0 \\
		&= 2\ln(3)(3) + 6(1) \\
		&= 6\ln(3) + 6.
	\end{align*}
	Therefore, $f'(0) = 6\ln(3) + 6$.
	\end{enumerate}
\end{solutionbox}

\subsection{Type 2.4: Sum of Inverse Trigonometric Functions}
\begin{problembox}{Problem 2.4: Inverse Trig Derivatives}
Given $f(x) = 3\arccos(4x) + 2\operatorname{arccot}(x)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\arccos(u)]' &= -\frac{1}{\sqrt{1-u^2}} u' \\
		[\operatorname{arccot}(u)]' &= -\frac{1}{1+u^2} u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left(3\arccos(4x) + 2\operatorname{arccot}(x)\right) \\
		&= \frac{d}{dx} 3\arccos(4x) + \frac{d}{dx} 2\operatorname{arccot}(x) \\
		&= 3 \frac{d}{dx} \arccos(4x) + 2 \frac{d}{dx} \operatorname{arccot}(x) \\
		&= 3\left(-\frac{1}{\sqrt{1-(4x)^2}}\right)\frac{d}{dx}(4x) + 2\left(-\frac{1}{1+x^2}\right)\frac{d}{dx}(x) \\
		&= (3)\left(-\frac{1}{\sqrt{1-(4x)^2}}\right)(4) + (2)\left(-\frac{1}{1+x^2}\right)(1).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = -\frac{12}{\sqrt{1-16x^2}} - \frac{2}{1+x^2}
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= -\frac{12}{\sqrt{1-16(0)^2}} - \frac{2}{1+(0)^2} \\
		&= -\frac{12}{\sqrt{1-0}} - \frac{2}{1+0} \\
		&= -\frac{12}{1} - \frac{2}{1} \\
		&= -12 - 2 \\
		&= -14.
	\end{align*}
	Therefore, $f'(0) = -14$.
	\end{enumerate}
\end{solutionbox}

\section{Product Rule: $(uv)' = uv' + vu'$}

\subsection{Type 3.1: Polynomial $\times$ Exponential}
\begin{problembox}{Problem 3.1: Product Rule (Poly $\times$ Exp)}
Given $f(x) = (2x + 3)e^{2x}$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		(u \cdot v)' &= u v' + v u' \\
		[e^u]' &= e^u u' \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} (2x + 3)e^{2x} \\
		&= (2x+3) \frac{d}{dx} e^{2x} + e^{2x} \frac{d}{dx} (2x + 3) \\
		&= (2x+3) e^{2x} \frac{d}{dx} 2x + e^{2x} \left[ \frac{d}{dx} 2x + \frac{d}{dx} 3 \right] \\
		&= (2x+3) e^{2x} (2) + e^{2x} \left[ 2 + 0 \right]
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = (4x + 6) e^{2x} + 2e^{2x}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= (4(0) + 6) e^{2(0)} + 2e^{2(0)} \\
		&= (0+6) e^{0} + 2e^{0} \\
		&= 6(1) + 2(1) \\
		&= 6+2 \\
		&= 8.
	\end{align*}
	Therefore, $f'(0) = 8$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 3.2: Exponential $\times$ Trigonometric (Damped Oscillation)}
\begin{problembox}{Problem 3.2: Product Rule (Exp $\times$ Trig)}
Given $f(x) = e^{-2x}\cos(3x)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		(u \cdot v)' &= u v' + v u' \\
		[e^u]' &= e^u u' \\
		[\cos(u)]' &= -\sin(u) u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ e^{-2x}\cos(3x) \right] \\
		&= e^{-2x} \frac{d}{dx} \cos(3x) + \cos(3x) \frac{d}{dx} e^{-2x} \\
		&= e^{-2x} \left( -\sin(3x) \frac{d}{dx}(3x) \right) + \cos(3x) \left( e^{-2x} \frac{d}{dx}(-2x) \right) \\
		&= e^{-2x} ( -\sin(3x) (3) ) + \cos(3x) ( e^{-2x} (-2) ) \\
		&= -3e^{-2x}\sin(3x) - 2e^{-2x}\cos(3x).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = -e^{-2x}(3\sin(3x) + 2\cos(3x)).
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= -e^{-2(0)}(3\sin(3(0)) + 2\cos(3(0))) \\
		&= -e^{0}(3\sin(0) + 2\cos(0)) \\
		&= -(1)(3(0) + 2(1)) \\
		&= -2.
	\end{align*}
	Therefore, $f'(0) = -2$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 3.3: Polynomial / Exponential $\times$ Inverse Trigonometric}
\begin{problembox}{Problem 3.3: Product Rule with Inverse Trig}
Given $f(x) = (e^{3x} + 2x)\arctan(x)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		(u \cdot v)' &= u v' + v u' \\
		[\arctan(u)]' &= \frac{1}{1+u^2} u' \\
		[e^u]' &= e^u u' \\
		[u^n]' &= n u^{n-1} u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ (e^{3x} + 2x)\arctan(x) \right] \\
		&= (e^{3x} + 2x) \frac{d}{dx} \arctan(x) + \arctan(x) \frac{d}{dx} (e^{3x} + 2x) \\
		&= (e^{3x} + 2x) \left( \frac{1}{1+x^2} \frac{d}{dx}(x) \right) + \arctan(x) \left[ \frac{d}{dx}(e^{3x}) + \frac{d}{dx}(2x) \right] \\
		&= (e^{3x} + 2x) \left( \frac{1}{1+x^2} (1) \right) + \arctan(x) \left[ e^{3x}\frac{d}{dx}(3x) + 2 \right] \\
		&= \frac{e^{3x} + 2x}{1+x^2} + \arctan(x) (3e^{3x} + 2).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{e^{3x} + 2x}{1+x^2} + (3e^{3x} + 2)\arctan(x).
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= \frac{e^{3(0)} + 2(0)}{1+(0)^2} + (3e^{3(0)} + 2)\arctan(0) \\
		&= \frac{e^{0} + 0}{1+0} + (3e^{0} + 2)(0) \\
		&= \frac{1 + 0}{1} + 0 \\
		&= 1.
	\end{align*}
	Therefore, $f'(0) = 1$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 4.1: Rational Polynomial Function}
\begin{problembox}{Problem 4.1: Quotient Rule (Polynomials)}
Given $f(x) = \frac{2x + 3}{x - 4}$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		\left(\frac{u}{v}\right)' &= \frac{v u' - u v'}{v^2} \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ \frac{2x + 3}{x - 4} \right] \\
		&= \frac{(x - 4) \frac{d}{dx}(2x + 3) - (2x + 3) \frac{d}{dx}(x - 4)}{(x - 4)^2} \\
		&= \frac{(x - 4) \left[ \frac{d}{dx}(2x) + \frac{d}{dx}(3) \right] - (2x + 3) \left[ \frac{d}{dx}(x) - \frac{d}{dx}(4) \right]}{(x - 4)^2} \\
		&= \frac{(x - 4)(2 + 0) - (2x + 3)(1 - 0)}{(x - 4)^2} \\
		&= \frac{2(x - 4) - 1(2x + 3)}{(x - 4)^2} \\
		&= \frac{2x - 8 - 2x - 3}{(x - 4)^2}.
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = -\frac{11}{(x - 4)^2}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= -\frac{11}{(0 - 4)^2} \\
		&= -\frac{11}{(-4)^2} \\
		&= -\frac{11}{16}.
	\end{align*}
	Therefore, $f'(0) = -\frac{11}{16}$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 4.2: Trigonometric / Radical Quotient}
\begin{problembox}{Problem 4.2: Quotient Rule (Trig / Radical)}
Given $f(x) = \frac{\sin(2x)}{\sqrt{x^2 + 1}}$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		\left(\frac{u}{v}\right)' &= \frac{v u' - u v'}{v^2} \\
		[\sin(u)]' &= \cos(u) u' \\
		[\sqrt{u}]' &= \frac{1}{2\sqrt{u}} u' \\
		[u^n]' &= n u^{n-1} u' \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ \frac{\sin(2x)}{\sqrt{x^2 + 1}} \right] \\
		&= \frac{\sqrt{x^2 + 1} \frac{d}{dx}\sin(2x) - \sin(2x) \frac{d}{dx}\sqrt{x^2 + 1}}{(\sqrt{x^2 + 1})^2} \\
		&= \frac{\sqrt{x^2 + 1} \left(\cos(2x)\frac{d}{dx}(2x)\right) - \sin(2x) \left(\frac{1}{2\sqrt{x^2+1}}\frac{d}{dx}(x^2+1)\right)}{x^2 + 1} \\
		&= \frac{\sqrt{x^2 + 1} \left(\cos(2x)(2)\right) - \sin(2x) \left(\frac{1}{2\sqrt{x^2+1}} \left[ \frac{d}{dx}(x^2) + \frac{d}{dx}(1) \right] \right)}{x^2 + 1} \\
		&= \frac{\sqrt{x^2 + 1} (2\cos(2x)) - \sin(2x) \left(\frac{1}{2\sqrt{x^2+1}} (2x + 0)\right)}{x^2 + 1} \\
		&= \frac{2\sqrt{x^2 + 1}\cos(2x) - \frac{x\sin(2x)}{\sqrt{x^2+1}}}{x^2 + 1}.
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{2(x^2+1)\cos(2x) - x\sin(2x)}{(x^2+1)^{\frac{3}{2}}}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= \frac{2(0^2+1)\cos(2(0)) - (0)\sin(2(0))}{(0^2+1)^{\frac{3}{2}}} \\
		&= \frac{2(1)\cos(0) - 0}{(1)^{\frac{3}{2}}} \\
		&= \frac{2(1) - 0}{1} \\
		&= 2.
	\end{align*}
	Therefore, $f'(0) = 2$.
\end{enumerate}
\end{solutionbox}

\section{Chain Rule \& Composite Functions}

\subsection{Type 5.1: Radical with Quadratic Inside $\sqrt{u(x)}$}
\begin{problembox}{Problem 5.1: Chain Rule with Radical}
Given $f(x) = \sqrt{2x^2 + 3x + 4}$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\sqrt{u}]' &= \frac{1}{2\sqrt{u}} u' \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \sqrt{2x^2 + 3x + 4} \\
		&= \frac{1}{2\sqrt{2x^2 + 3x + 4}} \frac{d}{dx} (2x^2 + 3x + 4) \\
		&= \frac{1}{2\sqrt{2x^2 + 3x + 4}} \left[ \frac{d}{dx}(2x^2) + \frac{d}{dx}(3x) + \frac{d}{dx}(4) \right] \\
		&= \frac{1}{2\sqrt{2x^2 + 3x + 4}} (4x + 3 + 0).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{4x + 3}{2\sqrt{2x^2 + 3x + 4}}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= \frac{4(0) + 3}{2\sqrt{2(0)^2 + 3(0) + 4}} \\
		&= \frac{3}{2\sqrt{0 + 0 + 4}} \\
		&= \frac{3}{2\sqrt{4}} \\
		&= \frac{3}{4}.
	\end{align*}
	Therefore, $f'(0) = \frac{3}{4}$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.2: Logarithmic with Polynomial Inside $\ln(u(x))$}
\begin{problembox}{Problem 5.2: Chain Rule with Natural Log}
Given $f(x) = \ln(x^2 + 3x + 2)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\ln(u)]' &= \frac{1}{u} u' \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \ln(x^2 + 3x + 2) \\
		&= \frac{1}{x^2 + 3x + 2} \frac{d}{dx} (x^2 + 3x + 2) \\
		&= \frac{1}{x^2 + 3x + 2} \left[ \frac{d}{dx}(x^2) + \frac{d}{dx}(3x) + \frac{d}{dx}(2) \right] \\
		&= \frac{1}{x^2 + 3x + 2} (2x + 3 + 0).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{2x + 3}{x^2 + 3x + 2}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= \frac{2(0) + 3}{(0)^2 + 3(0) + 2} \\
		&= \frac{3}{0 + 0 + 2} \\
		&= \frac{3}{2}.
	\end{align*}
	Therefore, $f'(0) = \frac{3}{2}$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.3: Trigonometric with Linear / Quadratic Inside}
\begin{problembox}{Problem 5.3: Chain Rule with Cotangent}
Given $f(x) = 2\cot\left(3x + \frac{\pi}{4}\right)$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\cot(u)]' &= -\csc^2(u) u' \\
		[u^n]' &= n u^{n-1} u' \\
		[c]' &= 0 \\
		(k_1f \pm k_2 g)' &= k_1f' \pm k_2g'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ 2\cot\left(3x + \frac{\pi}{4}\right) \right] \\
		&= 2 \left( -\csc^2\left(3x + \frac{\pi}{4}\right) \right) \frac{d}{dx} \left(3x + \frac{\pi}{4}\right) \\
		&= -2\csc^2\left(3x + \frac{\pi}{4}\right) \left[ \frac{d}{dx}(3x) + \frac{d}{dx}\left(\frac{\pi}{4}\right) \right] \\
		&= -2\csc^2\left(3x + \frac{\pi}{4}\right) (3 + 0).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = -6\csc^2\left(3x + \frac{\pi}{4}\right).
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= -6\csc^2\left(3(0) + \frac{\pi}{4}\right) \\
		&= -6\csc^2\left(\frac{\pi}{4}\right) \\
		&= -6(\sqrt{2})^2 \\
		&= -6(2) \\
		&= -12.
	\end{align*}
	Therefore, $f'(0) = -12$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.4: Inverse Trigonometric with Chain Rule}
\begin{problembox}{Problem 5.4: Chain Rule with ArcSec}
Given $f(x) = 2\operatorname{arcsec}(3x)$, find $f'(x)$ and evaluate $f'(1)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\operatorname{arcsec}(u)]' &= \frac{1}{|u|\sqrt{u^2-1}} u' \\
		[u^n]' &= n u^{n-1} u' \\
		(k_1f)' &= k_1f'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \left[ 2\operatorname{arcsec}(3x) \right] \\
		&= 2 \left( \frac{1}{|3x|\sqrt{(3x)^2-1}} \right) \frac{d}{dx} (3x) \\
		&= \frac{2}{|3x|\sqrt{9x^2-1}} (3).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{6}{|3x|\sqrt{9x^2-1}}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(1)$ by putting $x = 1$ into $f'(x)$, then we have
	\begin{align*}
		f'(1) &= \frac{6}{|3(1)|\sqrt{9(1)^2-1}} \\
		&= \frac{6}{3\sqrt{8}} \\
		&= \frac{2}{2\sqrt{2}} \\
		&= \frac{1}{\sqrt{2}} = \frac{\sqrt{2}}{2}.
	\end{align*}
	Therefore, $f'(1) = \frac{\sqrt{2}}{2}$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.5: Nested Composite Function ($\text{Trig}(\text{Inverse Trig}(kx))$)}
\begin{problembox}{Problem 5.5: Nested Trig of Inverse Trig}
Given $f(x) = \tan(\arcsin(3x))$, find $f'(x)$ and evaluate $f'(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		[\tan(u)]' &= \sec^2(u) u' \\
		[\arcsin(u)]' &= \frac{1}{\sqrt{1-u^2}} u' \\
		[u^n]' &= n u^{n-1} u'
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		f'(x) &= \frac{d}{dx} \tan(\arcsin(3x)) \\
		&= \sec^2(\arcsin(3x)) \frac{d}{dx} \arcsin(3x) \\
		&= \sec^2(\arcsin(3x)) \left( \frac{1}{\sqrt{1-(3x)^2}} \right) \frac{d}{dx} (3x) \\
		&= \sec^2(\arcsin(3x)) \left( \frac{1}{\sqrt{1-9x^2}} \right) (3).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = \frac{3\sec^2(\arcsin(3x))}{\sqrt{1-9x^2}}.
	\end{equation*}
	
	\item [Step 3.] Compute $f'(0)$ by putting $x = 0$ into $f'(x)$, then we have
	\begin{align*}
		f'(0) &= \frac{3\sec^2(\arcsin(3(0)))}{\sqrt{1-9(0)^2}} \\
		&= \frac{3\sec^2(\arcsin(0))}{\sqrt{1}} \\
		&= \frac{3\sec^2(0)}{1} \\
		&= 3(1)^2 \\
		&= 3.
	\end{align*}
	Therefore, $f'(0) = 3$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.6: Logarithmic Differentiation ($y = x^{kx}$)}
\begin{problembox}{Problem 5.6: Logarithmic Differentiation}
Given $f(x) = x^{3x}$ for $x > 0$, find $f'(x)$ and evaluate $f'(1)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		\ln(u^v) &= v \ln(u) \\
		[\ln(y)]' &= \frac{1}{y} y' \\
		(u \cdot v)' &= u v' + v u' \\
		[\ln(x)]' &= \frac{1}{x}
	\end{align*}
	
	\item [Step 2.] Find the derivative of $f$, we obtain
	\begin{align*}
		y &= x^{3x} \\
		\ln(y) &= \ln(x^{3x}) = 3x \ln(x) \\
		\frac{d}{dx} \ln(y) &= \frac{d}{dx} (3x \ln(x)) \\
		\frac{1}{y} \frac{dy}{dx} &= 3x \frac{d}{dx}\ln(x) + \ln(x) \frac{d}{dx}(3x) \\
		\frac{1}{y} \frac{dy}{dx} &= 3x \left(\frac{1}{x}\right) + \ln(x) (3) \\
		\frac{1}{y} \frac{dy}{dx} &= 3 + 3\ln(x) \\
		\frac{dy}{dx} &= y (3 + 3\ln(x)).
	\end{align*}
	Thus,
	\begin{equation*}
		f'(x) = x^{3x} (3 + 3\ln(x)).
	\end{equation*}
	
	\item [Step 3.] Compute $f'(1)$ by putting $x = 1$ into $f'(x)$, then we have
	\begin{align*}
		f'(1) &= (1)^{3(1)} (3 + 3\ln(1)) \\
		&= 1 (3 + 3(0)) \\
		&= 1(3) \\
		&= 3.
	\end{align*}
	Therefore, $f'(1) = 3$.
\end{enumerate}
\end{solutionbox}

\subsection{Type 5.7: Higher-Order Derivative ($f''(x)$)}
\begin{problembox}{Problem 5.7: Second Derivative}
Given $f(x) = x^2 e^{2x}$, find $f''(x)$ and evaluate $f''(0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		(u \cdot v)' &= u v' + v u' \\
		[e^u]' &= e^u u' \\
		[u^n]' &= n u^{n-1} u'
	\end{align*}
	
	\item [Step 2.] Find the first derivative $f'(x)$:
	\begin{align*}
		f'(x) &= \frac{d}{dx} (x^2 e^{2x}) \\
		&= x^2 \frac{d}{dx}e^{2x} + e^{2x} \frac{d}{dx}x^2 \\
		&= x^2(e^{2x}(2)) + e^{2x}(2x) \\
		&= 2x^2 e^{2x} + 2x e^{2x}.
	\end{align*}
	Find the second derivative $f''(x)$:
	\begin{align*}
		f''(x) &= \frac{d}{dx} (2x^2 e^{2x} + 2x e^{2x}) \\
		&= \frac{d}{dx} (2x^2 e^{2x}) + \frac{d}{dx} (2x e^{2x}) \\
		&= \left[ 2x^2 \frac{d}{dx}e^{2x} + e^{2x} \frac{d}{dx}2x^2 \right] + \left[ 2x \frac{d}{dx}e^{2x} + e^{2x} \frac{d}{dx}2x \right] \\
		&= \left[ 2x^2\left(e^{2x}\frac{d}{dx}(2x)\right) + e^{2x}(4x) \right] + \left[ 2x\left(e^{2x}\frac{d}{dx}(2x)\right) + e^{2x}(2) \right] \\
		&= \left[ 2x^2(e^{2x}(2)) + e^{2x}(4x) \right] + \left[ 2x(e^{2x}(2)) + e^{2x}(2) \right] \\
		&= 4x^2 e^{2x} + 4x e^{2x} + 4x e^{2x} + 2 e^{2x}.
	\end{align*}
	Thus,
	\begin{equation*}
		f''(x) = (4x^2 + 8x + 2) e^{2x}.
	\end{equation*}
	
	\item [Step 3.] Compute $f''(0)$ by putting $x = 0$ into $f''(x)$, then we have
	\begin{align*}
		f''(0) &= (4(0)^2 + 8(0) + 2) e^{2(0)} \\
		&= (0 + 0 + 2) e^{0} \\
		&= 2(1) \\
		&= 2.
	\end{align*}
	Therefore, $f''(0) = 2$.
\end{enumerate}
\end{solutionbox}

\section{Implicit Differentiation}

\subsection{Type 6.1: Circle Equation ($x^2 + y^2 = r^2$)}
\begin{problembox}{Problem 6.1: Implicit Differentiation of a Circle}
Find $\frac{dy}{dx}$ for the equation $x^2 + y^2 = 25$ at the point $(3, 4)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		\frac{d}{dx}[x^n] &= n x^{n-1} \\
		\frac{d}{dx}[y^n] &= n y^{n-1} \frac{dy}{dx} \\
		\frac{d}{dx}[c] &= 0
	\end{align*}
	
	\item [Step 2.] Differentiate both sides of the equation with respect to $x$:
	\begin{align*}\label{eq;6.1.1}
		\frac{d}{dx}\left[x^2 + y^2\right] &= \frac{d}{dx}[25] \\
		\frac{d}{dx}[x^2] + \frac{d}{dx}[y^2] &= 0.
	\end{align*}

	\item [Step 3.] Calculate each derivative term:
	\begin{align*}
		\frac{d}{dx}[x^2] &= 2x\frac{dx}{dx} = 2x(1) = 2x \\
		\frac{d}{dx}[y^2] &= 2y\frac{dy}{dx}
	\end{align*}

	\item [Step 4.] Substitute back and isolate $\frac{dy}{dx}$:
	\begin{align*}
		2x + 2y\frac{dy}{dx} &= 0 \\
		2y\frac{dy}{dx} &= -2x \\
		\frac{dy}{dx} &= -\frac{2x}{2y} = -\frac{x}{y}.
	\end{align*}

	\item [Step 5.] Substitute the given point $(x, y) = (3, 4)$:
	\begin{equation*}
		\left.\frac{dy}{dx}\right|_{(3, 4)} = -\frac{3}{4}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 6.2: Mixed Polynomial with $xy$ Product Term}
\begin{problembox}{Problem 6.2: Implicit Differentiation with Product Rule}
Find $\frac{dy}{dx}$ for the equation $x^3 + y^3 - 3xy = 5$ at the point $(x_0, y_0)$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Rules and formula used in this problem are:
	\begin{align*}
		\frac{d}{dx}[u \cdot v] &= u \frac{dv}{dx} + v \frac{du}{dx} \\
		\frac{d}{dx}[y^n] &= n y^{n-1} \frac{dy}{dx} \\
		\frac{d}{dx}[x^n] &= n x^{n-1} \\
		\frac{d}{dx}[c] &= 0
	\end{align*}
	
	\item [Step 2.] Differentiate both sides of the equation with respect to $x$:
	\begin{align*}\label{eq;6.2.1}
		\frac{d}{dx}\left[x^3 + y^3 - 3xy\right] &= \frac{d}{dx}[5] \\
		\frac{d}{dx}[x^3] + \frac{d}{dx}[y^3] - 3\frac{d}{dx}[xy] &= 0.
	\end{align*}

	\item [Step 3.] Calculate each derivative term:
	\begin{align*}
		\frac{d}{dx}[x^3] &= 3x^2\frac{dx}{dx} = 3x^2(1) = 3x^2 \\
		\frac{d}{dx}[y^3] &= 3y^2\frac{dy}{dx} \\
		\frac{d}{dx}[xy] &= x\frac{dy}{dx} + y\frac{dx}{dx} = x\frac{dy}{dx} + y(1) = x\frac{dy}{dx} + y
	\end{align*}

	\item [Step 4.] Combine and isolate $\frac{dy}{dx}$:
	\begin{align*}
		3x^2 + 3y^2\frac{dy}{dx} - 3\left(x\frac{dy}{dx} + y\right) &= 0 \\
		3x^2 + 3y^2\frac{dy}{dx} - 3x\frac{dy}{dx} - 3y &= 0 \\
		(3y^2 - 3x)\frac{dy}{dx} &= 3y - 3x^2 \\
		\frac{dy}{dx} &= \frac{3y - 3x^2}{3y^2 - 3x} = \frac{y - x^2}{y^2 - x}.
	\end{align*}

	\item [Step 5.] Evaluate at the given point $(x_0, y_0)$, we obtain:
	\begin{equation*}
		\left.\frac{dy}{dx}\right|_{(x_0, y_0)} = \frac{y_0 - x_0^2}{y_0^2 - x_0}.
	\end{equation*}
\end{enumerate}
\end{solutionbox}

\section{Tangent \& Normal Lines}

\subsection{Type 7.1: Equation of the Tangent Line}
\begin{problembox}{Problem 7.1: Tangent Line Equation}
Find the slope of the tangent line $m_t$ and the equation of the tangent line to the curve $y = x^2 - 3x + 2$ at $x = 2$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Formulas for the tangent line:
	\begin{align*}
		m_t &= f'(x_0) \\
		y - y_0 &= m_t(x - x_0)
	\end{align*}
	
	\item [Step 2.] Find the derivative $f'(x)$:
	\begin{equation}\label{eq;7.1.1}
		f'(x) = \frac{d}{dx}\left[x^2 - 3x + 2\right] = 2x - 3.
	\end{equation}

	\item [Step 3.] Find the point of tangency $(x_0, y_0)$ by evaluating $y(2)$:
	\begin{equation*}
		y_0 = f(2) = (2)^2 - 3(2) + 2 = 4 - 6 + 2 = 0
	\end{equation*}
	implies $(x_0, y_0) = (2, 0)$.

	\item [Step 4.] Find the slope of the tangent line $m_t$:
	\begin{equation*}
		m_t = f'(2) = 2(2) - 3 = 4 - 3 = 1.
	\end{equation*}

	\item [Step 5.] Construct the equation of the tangent line:
	\begin{align*}
		y - y_0 &= m_t(x - x_0) \\
		y - 0 &= 1(x - 2) \\
		y &= x - 2 \quad \left(\text{or } x - y - 2 = 0\right).
	\end{align*}
\end{enumerate}
\end{solutionbox}

\subsection{Type 7.2: Equation of the Normal Line}
\begin{problembox}{Problem 7.2: Normal Line Equation}
Find the slope of the normal line $m_n$ and the equation of the normal line to the curve $y = x^2 - 3x + 2$ at $x = 2$.
\end{problembox}
\begin{solutionbox}
\begin{enumerate}
	\item [Step 1.] Formulas for the normal line:
	\begin{align*}
		m_t &= f'(x_0) \\
		m_n &= -\frac{1}{m_t} \quad (m_t \neq 0) \\
		y - y_0 &= m_n(x - x_0)
	\end{align*}
	
	\item [Step 2.] Find the derivative $f'(x)$:
	\begin{equation}\label{eq;7.2.1}
		f'(x) = \frac{d}{dx}\left[x^2 - 3x + 2\right] = 2x - 3.
	\end{equation}

	\item [Step 3.] Find the point $(x_0, y_0)$ by evaluating $y(2)$:
	\begin{equation*}
		y_0 = f(2) = (2)^2 - 3(2) + 2 = 4 - 6 + 2 = 0
	\end{equation*}
	implies $(x_0, y_0) = (2, 0)$.

	\item [Step 4.] Find the slope of the normal line $m_n$:
	\begin{align*}
		m_t &= f'(2) = 2(2) - 3 = 1 \\
		m_n &= -\frac{1}{m_t} = -\frac{1}{1} = -1.
	\end{align*}

	\item [Step 5.] Construct the equation of the normal line:
	\begin{align*}
		y - y_0 &= m_n(x - x_0) \\
		y - 0 &= -1(x - 2) \\
		y &= -x + 2 \quad \left(\text{or } x + y - 2 = 0\right).
	\end{align*}
\end{enumerate}
\end{solutionbox}
